#lang racket/base
(require "../code/id_q4.rkt")
(require racket/port)

;; Test cases: each is (x lst n expected)
(define test-cases
  (list
    (list 'a '(1 2) 3 '(a 1 2))
    (list 'x '() 1 '(x))
    (list 'z '((1) (2 3)) 2 '(z (z 1) (z 2 3)))
    (list #\\a '((( )) ) 4 '(a (a (a ()))) )
    (list 42 '(1 2 3 4) 5 '(42 1 2 3 4))
    (list 'y '(1 (2 (3 (4)))) 2 '(y 1 (y 2 (y 3 (y 4))))) ))

(define (run-tests)
  (for ([tc (in-list test-cases)]
        [i (in-naturals 1)])
    (define x (list-ref tc 0))
    (define lst (list-ref tc 1))
    (define n (list-ref tc 2))
    (define expected (list-ref tc 3))
    (define input-str (format "~a ~s ~a" x (format "~s" lst) n))
    (define result
      (with-output-to-string
        (lambda ()
          (with-input-from-string input-str
            (lambda () (print-insert-result))))) )
    (define got (read (open-input-string result)))
    (if (equal? got expected)
        (printf "Test ~a passed.\n" i)
        (printf "Test ~a FAILED. Expected ~s, got ~s\n" i expected got))))

(run-tests)
